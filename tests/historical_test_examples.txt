--- PowerShell Batch Script (Windows) ---
Save as: run_historical_tests.ps1

$cases = @(
  @{ mun="Arteaga";              est="Coahuila";         date="2021-03-16" }, # 3,630 ha
  @{ mun="Zapopan";              est="Jalisco";          date="2021-03-31" }, # 4,872 ha
  @{ mun="Ensenada";             est="Baja California";  date="2019-10-24" }, # 21,795 ha
  @{ mun="San Miguel Chimalapa"; est="Oaxaca";           date="2019-04-19" }, # 20,350 ha
  @{ mun="Guadalupe y Calvo";    est="Chihuahua";        date="2023-05-18" }, # 5,238 ha
  @{ mun="Pueblo Nuevo";         est="Durango";          date="2021-04-15" }, # 9,300 ha
  @{ mun="Uruapan";              est="Michoacan";        date="2022-05-09" }, # 2,697 ha
  @{ mun="Tepoztlan";            est="Morelos";          date="2025-04-09" }, # 1,380 ha
  @{ mun="Perote";               est="Veracruz";         date="2019-05-08" }, # 1,045 ha
  @{ mun="Valle de Bravo";       est="State of Mexico";  date="2021-04-20" }, # 996 ha
  @{ mun="Villaflores";          est="Chiapas";          date="2024-04-02" }, # 3,865 ha
  @{ mun="Bacalar";              est="Quintana Roo";     date="2024-05-18" }  # 13,671 ha
)

Write-Host "Running Historical Wildfire Verification Tests..." -ForegroundColor Cyan

foreach ($c in $cases) {
    Write-Host "Querying $($c.mun), $($c.est) on $($c.date)..." -ForegroundColor Yellow
    $body = $c | ConvertTo-Json
    $res = Invoke-RestMethod -Uri "http://127.0.0.1:5000/api/predict" -Method Post -Body $body -ContentType "application/json"
    
    $prob = $res.probability
    $risk = $res.risk_level
    $tmax = $res.weather.temp_max_c
    $rh   = $res.weather.relative_humidity_pct
    $dry  = $res.weather.consecutive_dry_days
    
    Write-Host "  -> Probability: $prob% | Risk: $risk | Tmax: ${tmax}°C | Humidity: ${rh}% | Dry days: $dry" -ForegroundColor Green
}

--- Bash Script (Linux / macOS / WSL) ---

#!/bin/bash
TEST_CASES=(
  '{"municipio":"Arteaga","estado":"Coahuila","fecha":"2021-03-16"}'
  '{"municipio":"Zapopan","estado":"Jalisco","fecha":"2021-03-31"}'
  '{"municipio":"Ensenada","estado":"Baja California","fecha":"2019-10-24"}'
  '{"municipio":"San Miguel Chimalapa","estado":"Oaxaca","fecha":"2019-04-19"}'
  '{"municipio":"Guadalupe y Calvo","estado":"Chihuahua","fecha":"2023-05-18"}'
  '{"municipio":"Pueblo Nuevo","estado":"Durango","fecha":"2021-04-15"}'
  '{"municipio":"Uruapan","estado":"Michoacan","fecha":"2022-05-09"}'
  '{"municipio":"Tepoztlan","estado":"Morelos","fecha":"2025-04-09"}'
  '{"municipio":"Perote","estado":"Veracruz","fecha":"2019-05-08"}'
  '{"municipio":"Valle de Bravo","estado":"State of Mexico","fecha":"2021-04-20"}'
  '{"municipio":"Villaflores","estado":"Chiapas","fecha":"2024-04-02"}'
  '{"municipio":"Bacalar","estado":"Quintana Roo","fecha":"2024-05-18"}'
)

for payload in "${TEST_CASES[@]}"; do
  echo "Testing historical fire: $payload"
  curl -s -X POST http://127.0.0.1:5000/api/predict \
       -H "Content-Type: application/json" \
       -d "$payload" | jq '{municipio, estado, date, probability, risk_level, weather: {temp_max_c: .weather.temp_max_c, relative_humidity_pct: .weather.relative_humidity_pct, consecutive_dry_days: .weather.consecutive_dry_days}}'
  echo "----------------------------------------------------"
done
