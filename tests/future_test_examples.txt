$cases = @(
  @{ mun="Ensenada";           est="Baja California";  date="2026-09-05" },
  @{ mun="Guadalupe y Calvo";  est="Chihuahua";        date="2026-09-07" },
  @{ mun="Zapopan";            est="Jalisco";          date="2026-09-10" },
  @{ mun="Tlalpan";            est="Mexico City";      date="2026-09-11" },
  @{ mun="Uruapan";            est="Michoacan";        date="2026-09-12" },
  @{ mun="Calvillo";           est="Aguascalientes";   date="2026-09-13" },
  @{ mun="Valle de Bravo";     est="State of Mexico";  date="2026-09-14" },
  @{ mun="Perote";             est="Veracruz";         date="2026-09-15" },
  @{ mun="Oaxaca de Juarez";   est="Oaxaca";           date="2026-09-16" },
  @{ mun="Othon P. Blanco";    est="Quintana Roo";     date="2026-09-18" },
  @{ mun="Calvillo";           est="Aguascalientes";   date="2026-09-19" } # Edge case check
)

foreach ($c in $cases) {
    Write-Host "Testing $($c.mun), $($c.est) for $($c.date)..." -ForegroundColor Cyan
    $body = $c | ConvertTo-Json
    $res = Invoke-RestMethod -Uri "http://127.0.0.1:5000/api/predict" -Method Post -Body $body -ContentType "application/json"
    if ($res.warning) {
        Write-Host "  -> Warning/Boundary: $($res.warning)" -ForegroundColor Yellow
    } else {
        Write-Host "  -> Probability: $($res.probability)% | Risk: $($res.risk_level) | Source: $($res.data_source)" -ForegroundColor Green
    }
}
